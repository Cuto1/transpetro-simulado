function preferredDistinct(items){

 for(const q of items){
   const root=q.duplicateOf||q.id;
   const prev=groups.get(root);

 }
 return [...groups.values()];
}

function subjectStats(){
  const groups={};
  for(const q of QUESTIONS){
    const name=q.planTheme||q.theme||"Outros";
    if(!groups[name])groups[name]={name,n:0,c:0,available:0,wrong:[],unanswered:[],attempts:0,practiceCorrect:0};
    const g=groups[name], a=state.answers[q.id];
    if(a){g.n++;if(a.correct)g.c++;else g.wrong.push(q)}else g.unanswered.push(q);
    const hist=state.reinforcement[q.id]||[];
    g.attempts+=hist.length;g.practiceCorrect+=hist.filter(x=>x.correct).length;
    g.available++;
  }
  return Object.values(groups).map(g=>({...g,rate:g.n?g.c/g.n:null,level:g.n<5?"diagnostico":g.c/g.n<.6?"urgente":g.c/g.n<.75?"atencao":"manutencao"}))

}

  return out;
}
